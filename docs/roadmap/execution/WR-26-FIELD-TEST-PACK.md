## الاستعلامات (قراءةٌ فقط، تُشغَّلُ بـ`:order_id` و`:driver_id` و`:city_code`)

```sql
-- ق-0: لا سائقَ متوفّرًا في المدينةِ غيرُ سائقِ الاختبار
select d.id from drivers d join driver_availability a on a.driver_id = d.id
 join cities c on c.id = d.city_id
 where c.code = :city_code and a.is_available and d.id <> :driver_id;

-- ق-1: أهليّةُ السائقِ كما يراها التوزيع
select verification_status, last_location_at, now() - last_location_at as age,
       last_location_quality, last_location_accuracy_m
  from drivers where id = :driver_id;

-- ق-2: حالُ الطلب
select status, service, pickup_at, rider_offer_mode, rider_offer_sar, notes,
       candidate_driver_id, candidate_expires_at, assigned_driver_id,
       matched_at, arrived_at, started_at, completed_at, cancelled_reason
  from orders where id = :order_id;

-- ق-3: الصادرُ للطلب (لا تكرارَ لكلِّ مفتاح)
select kind, status, attempts, error_code, dead_reason, dedup_key, delivered_at
  from notification_outbox where order_id = :order_id order by created_at;

-- ق-4: العروض
select driver_id, round, status, responded_at, expires_at
  from order_offers where order_id = :order_id order by created_at;

-- ق-5: المحادثةُ بلا نصوص (الطولُ فقط)
select seq, sender_role, length(body) as len, created_at
  from order_messages where order_id = :order_id order by seq;

-- ق-6: التقييمات
select direction, stars, count(*) over (partition by direction) as n
  from ratings where order_id = :order_id;
```
